if(ANDROID_ABI STREQUAL "x86_64" AND SEED_BRIDGE_RENDERER)
  if(NOT EXISTS "${SEED_BRIDGE_RENDERER}")
    message(FATAL_ERROR "SEED_BRIDGE_RENDERER must be an existing immutable ARM64 renderer")
  endif()
  file(SIZE "${SEED_BRIDGE_RENDERER}" RENDERER_SIZE)
  if(RENDERER_SIZE LESS 20 OR RENDERER_SIZE GREATER 33554432)
    message(FATAL_ERROR "Bridge renderer size outside supported bounds")
  endif()
  file(READ "${SEED_BRIDGE_RENDERER}" RENDERER_HEADER LIMIT 20 HEX)
  string(SUBSTRING "${RENDERER_HEADER}" 0 14 RENDERER_FORMAT)
  string(SUBSTRING "${RENDERER_HEADER}" 32 8 RENDERER_MACHINE)
  if(NOT RENDERER_FORMAT STREQUAL "7f454c46020101" OR NOT RENDERER_MACHINE STREQUAL "0300b700")
    message(FATAL_ERROR "Bridge renderer must be a little-endian ARM64 ELF64 shared object")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SEED_BRIDGE_RENDERER}")
  file(SHA256 "${SEED_BRIDGE_RENDERER}" BRIDGE_RENDERER_SHA)
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/seed-bridge-renderer.sha256" CONTENT "${BRIDGE_RENDERER_SHA}\n" NEWLINE_STYLE UNIX)
  file(GLOB BRIDGE_XDL "${G}/xdl/*.c")
  # Bundled candidate; production x86_64 enable remains refused until qualified.
  add_library(seed_bridge SHARED bridge_host.cpp ${BRIDGE_XDL})
  target_include_directories(seed_bridge PRIVATE "${A}" "${G}/xdl/include")
  target_compile_definitions(seed_bridge PRIVATE SEED_BRIDGE_RENDERER_SHA="${BRIDGE_RENDERER_SHA}")
  if(SEED_API31_CANDIDATE)
    target_compile_definitions(seed_bridge PRIVATE SEED_API31_CANDIDATE)
  endif()
  if(SEED_API34_CANDIDATE)
    target_compile_definitions(seed_bridge PRIVATE SEED_API34_CANDIDATE)
  endif()
  set_source_files_properties(bridge_host.cpp PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
  target_compile_options(seed_bridge PRIVATE -O2 -fvisibility=hidden)
  target_link_options(seed_bridge PRIVATE -Wl,-z,max-page-size=16384)
  target_link_libraries(seed_bridge dl)
endif()
