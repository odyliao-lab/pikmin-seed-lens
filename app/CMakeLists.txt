cmake_minimum_required(VERSION 3.21)
project(seed_preview_root LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(A "${ROOT}/native/renderer")
set(G "${ROOT}/third_party")
set(K "${ROOT}/third_party/AndKittyInjector")
file(GLOB KM "${K}/KittyMemoryEx/KittyMemoryEx/*.cpp")
file(GLOB INJ "${K}/AndKittyInjector/src/Injector/*.cpp" "${K}/AndKittyInjector/src/Utils/*.cpp")
# Embed the library, NOT the upstream general-purpose CLI/watch/hide interface.
add_executable(seed-root loader.cpp ${KM} ${INJ})
target_include_directories(seed-root PRIVATE "${K}/AndKittyInjector/src" "${K}/KittyMemoryEx/KittyMemoryEx" "${A}")
target_compile_definitions(seed-root PRIVATE kNO_KEYSTONE NDEBUG)
target_compile_options(seed-root PRIVATE -O2 -ffunction-sections -fdata-sections)
target_link_options(seed-root PRIVATE -Wl,--gc-sections)
target_link_libraries(seed-root log dl)
if(ANDROID_ABI STREQUAL "arm64-v8a")
  file(GLOB XDL "${G}/xdl/*.c")
  add_library(seed_display SHARED payload.cpp "${A}/seed_preview.cpp" "${G}/And64InlineHook/And64InlineHook.cpp" ${XDL})
  target_include_directories(seed_display PRIVATE "${A}" "${G}/And64InlineHook" "${G}/xdl/include")
  target_compile_options(seed_display PRIVATE -O2 -fvisibility=hidden)
  target_link_libraries(seed_display log dl)
endif()
add_executable(seed_preview_policy_test "${ROOT}/native/tests/seed_preview_policy_test.cpp")
add_executable(identity_test identity_test.cpp)
