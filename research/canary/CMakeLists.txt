cmake_minimum_required(VERSION 3.21)
project(seed_canary_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT ANDROID_ABI STREQUAL "x86_64")
  message(FATAL_ERROR "Research host is x86_64 only")
endif()
set(K "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/AndKittyInjector")
execute_process(COMMAND git -C "${K}" rev-parse HEAD OUTPUT_VARIABLE INJ_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND git -C "${K}/KittyMemoryEx" rev-parse HEAD OUTPUT_VARIABLE KM_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND git -C "${K}" status --porcelain OUTPUT_VARIABLE DEP_DIRTY COMMAND_ERROR_IS_FATAL ANY)
if(NOT INJ_HEAD STREQUAL "9ece8bc0f05344ab23e882a38728ab91f70bfed0" OR
   NOT KM_HEAD STREQUAL "be4cdc9a79e3e1b93357e8d30608e84a3518f5d3" OR DEP_DIRTY)
  message(FATAL_ERROR "Pinned dependencies must match and remain clean")
endif()
file(GLOB KM "${K}/KittyMemoryEx/KittyMemoryEx/*.cpp")
file(GLOB INJ "${K}/AndKittyInjector/src/Injector/*.cpp" "${K}/AndKittyInjector/src/Utils/*.cpp")
# Isolated copy only; pinned submodule stays immutable. The parent watchdog is
# still mandatory: syscall recovery does not make arbitrary dlopen cancellable.
set(TRACE "${K}/KittyMemoryEx/KittyMemoryEx/KittyTrace.cpp")
include("${CMAKE_CURRENT_SOURCE_DIR}/../bounded-trace.cmake")
list(REMOVE_ITEM KM "${TRACE}")
add_executable(canary-root loader.cpp ${KM} ${INJ} "${CMAKE_CURRENT_BINARY_DIR}/KittyTrace-bounded.cpp")
target_include_directories(canary-root PRIVATE "${K}/AndKittyInjector/src" "${K}/KittyMemoryEx/KittyMemoryEx")
target_compile_definitions(canary-root PRIVATE kNO_KEYSTONE NDEBUG)
target_link_libraries(canary-root log dl)
file(GLOB XDL "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/xdl/*.c")
add_library(seedprobe_host SHARED host.cpp ${XDL})
if(NOT EXISTS "${SEED_CANARY_LIBRARY}")
  message(FATAL_ERROR "Set SEED_CANARY_LIBRARY to the immutable ARM64 canary packaged in the test APK")
endif()
file(SHA256 "${SEED_CANARY_LIBRARY}" CANARY_SHA)
target_compile_definitions(seedprobe_host PRIVATE SEED_CANARY_SHA="${CANARY_SHA}")
if(SEED_FOREIGN_APK)
  target_compile_definitions(seedprobe_host PRIVATE SEED_FOREIGN_APK="${SEED_FOREIGN_APK}")
endif()
target_include_directories(seedprobe_host PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/xdl/include"
  "${CMAKE_CURRENT_SOURCE_DIR}/../../app" "${CMAKE_CURRENT_SOURCE_DIR}/../../native/renderer")
set_source_files_properties(host.cpp PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
target_compile_options(seedprobe_host PRIVATE -fvisibility=hidden)
target_link_libraries(seedprobe_host dl)
add_executable(canary_identity_test identity_test.cpp)
add_executable(apk_payload_test apk_payload_test.cpp)
target_include_directories(apk_payload_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../app"
  "${CMAKE_CURRENT_SOURCE_DIR}/../../native/renderer")
target_compile_options(apk_payload_test PRIVATE -Wall -Wextra -Werror)
add_executable(kitty_syscall_test ../kitty_syscall_test.cpp ${KM} "${CMAKE_CURRENT_BINARY_DIR}/KittyTrace-bounded.cpp")
target_include_directories(kitty_syscall_test PRIVATE "${K}/KittyMemoryEx/KittyMemoryEx")
target_compile_definitions(kitty_syscall_test PRIVATE kNO_KEYSTONE NDEBUG)
target_link_libraries(kitty_syscall_test log dl)
add_executable(x64_call_context_test ../x64_call_context_test.cpp ${KM} "${CMAKE_CURRENT_BINARY_DIR}/KittyTrace-bounded.cpp")
target_include_directories(x64_call_context_test PRIVATE "${K}/KittyMemoryEx/KittyMemoryEx")
target_compile_definitions(x64_call_context_test PRIVATE kNO_KEYSTONE NDEBUG)
set_source_files_properties(../x64_call_context_test.cpp PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
target_link_libraries(x64_call_context_test log dl)
add_executable(game_entry_probe ../game_entry_probe.cpp)
target_compile_options(game_entry_probe PRIVATE -Wall -Wextra -Werror)
add_executable(entry_context_test ../entry_context_test.cpp)
target_compile_options(entry_context_test PRIVATE -Wall -Wextra -Werror)
